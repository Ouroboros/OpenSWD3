OpenSWD3軒轅劍參 雲和山的彼端 · 重制执行程序

《軒轅劍參 雲和山的彼端》（v1.02 DVD 版）的重制执行程序。它取代原版的 swd3.exe，读取你已安装的原版游戏资料，在现在的 Windows 上直接运行。

剧情、战斗、数值与各种规则都和原版一样，连原版的小 BUG 也保留；会让游戏崩溃的问题则已修正。

本项目不含任何游戏资料，需要你已有原版游戏。

安装与运行

  1. 准备好原版游戏目录，例如 E:\Game\swd3。
  2. 从 Releases 下载最新版的压缩包，解压后得到 OpenSWD3 文件夹，里面有 swd3r.exe、asset 文件夹和设置文件 swd3r.toml。可以整个放在任意位置，也可以把里面的东西放进游戏目录。
  3. 如果没有放进游戏目录，用记事本打开 swd3r.toml，在 [paths] 里写上游戏目录： 
    [paths]
data_dir = 'E:\Game\swd3'
  4. 运行 swd3r.exe。

设置

exe 旁边有两个设置文件：

  • swd3r.toml：你自己的设置。程序只读不改，注释和顺序都保留。
  • swd3r.default.toml：程序自动保存的设置，包括游戏里改的设置、按键和退出时的窗口大小。

同一项两个文件都写了时，以 swd3r.toml 为准，这时在游戏里再改只在当次有效。swd3r.toml 里每一项都带着说明，默认被 # 注释掉，删掉行首的 # 就会生效。值写错或没写时取默认值，写错的会在日志里记一条警告。

下面是 swd3r.toml 的例子，除了游戏目录，写的都是默认值：

[paths]
# 原版游戏目录。不写时用当前目录，所以把 exe 放进游戏目录就不用写。
# 相对路径以 exe 所在的目录为准。路径用单引号时反斜杠照写，用双引号时要写成 \\。
data_dir = 'E:\Game\swd3'
# 存档目录。不写时和原版一样存在游戏目录的 save 里。
save_dir = 'E:\Game\swd3\save'

[window]
# 窗口的宽和高，至少 640×480。画面保持原来的比例尽量填满窗口，比例不同时留黑边。
# 这三项不写时，用上次退出时的窗口大小。游戏里按 Alt+Enter 切换全屏。
width = 1280
height = 960
# 启动时是否最大化。
maximized = false

[display]
# 以下六项也能在游戏里改：标题画面的「環境設定」，或游戏中按 Esc →「設定」→「顯示設定」。

# 平滑畫面。游戏逻辑仍和原版一样每 35 毫秒走一步，移动、淡入淡出等动态在两步之间
# 按显示器刷新率连续画出，高刷新率显示器更流畅。false 时和原版一样每 35 毫秒画一次。
smooth_motion = true
# 垂直同步。等显示器刷新再显示，消除画面撕裂。
vsync = true
# 每秒最多画几帧，可写 30～1000；0 表示不限，开着垂直同步时就跟随显示器刷新率。
# 只在平滑畫面开启时有效。
frame_rate = 0
# 平滑畫面按几倍精度绘制，可写 1～8。倍数越高，镜头和人物移动越细，也越耗 CPU
# （2 倍约为 1 倍的 3～4 倍）。"auto" 按窗口大小取能完整显示的最大倍数，最多 4 倍。
# 只在平滑畫面开启时有效。
render_scale = "auto"
# 地图填满宽屏窗口，能看到更大范围，对话框仍在中间。
# 菜单、战斗、存档等画面仍是 640×480，两边留黑边。
wide_map = true
# 道具提示。地图上看不见、但调查后能拿到道具或银币的地方，显示一个跳动的箭头
# （包括鼠标移上去不变手形、只能面对它按键调查的地方），还没打开的罐子、宝箱也显示。
# 能调查时才显示，拿走后消失。
item_hints = false

[game]
# 启动时先显示启动画面；false 时直接进入标题画面。游戏里在「設定」第一行切换。
launcher = true
# 自動連擊。自动战斗时，戴着「連擊」饰品（武聖手鐲、姜公膽）且体力足够的角色自动使用「連擊」，
# 体力不够时照常普通攻击。游戏里在「設定」→「戰鬥設定」切换。
auto_combo = false
# 地图上默认跑步，false 是走路。游戏里按 R 切换时会记下来，下次启动照旧。
running = true
# 调试模式，见 README 的「调试模式」。
debug = false
# [game] 里还有音量、文字速度、战斗速度和按键（[game.keys]），和原版「環境設定」里的一样，
# 在游戏里改就行。第一次启动时，这些设置会从游戏目录的 Env.dat 导入一次。

[logging]
# 日志的位置和文件名。每次启动写一个新文件，文件名后面加上启动时间和进程号，
# 例如 logs/swd3r-2026-10-26_14-32-05-127840.log。相对路径以 exe 所在的目录为准。
path = "logs/swd3r.log"
# 记录多详细，从多到少是 trace、debug、info、warning、error、critical。
level = "debug"

角色属性

各项属性在原版里的实际作用（重制版的计算与原版相同）。属性都存成 16 位整数，最大 32767；表中的「上限」是超过后会被截掉、或再高也没有用的值。

属性作用上限
生命降到 0 就倒下9999。每场战斗开始时，参战角色的生命和生命上限超过 9999 的部分会被永久截掉。胜利升级时在 9999 上再加成长，可以暂时超过，到下一场战斗又被截回 9999
靈力施放法术和部分招式时消耗（每个招式自己规定扣灵力还是扣体力）9999，同生命
體力使用部分招式和饰品指令（連擊、投擲、召喚等）时消耗9999，同生命
力量力量加上装备的攻击力，就是状态页的「攻擊」。普通攻击的伤害 = 攻擊 + 随机 1～（等級 ÷ 3 + 3）− 敌人防禦，至少 1；攻击提升、攻击下降状态各增减 30%。投擲（暗殺寶錄、御夫錄）另外按「力量 × 倍数」加伤害，倍数随熟练度从 1 到 4单次伤害最多 9999，攻擊减去敌人防禦超过 9999 后再高也没用。攻擊超过约 32740（有攻击提升状态时约 25200）时计算溢出，伤害反而只剩 1
耐力耐力加上装备的防御力，就是状态页的「防禦」。只减少敌人普通攻击的伤害：伤害 = 敌人攻擊 + 随机 − 防禦；防禦不低于敌人攻擊时，只受 1 加随机 1～（敌人等級 ÷ 3 + 3）的伤害。法术伤害不看防禦没有截断；高过敌人的攻擊后再高也没用
智慧法术等按招式数值计算的伤害，额外加随机 1～（等級 + 智慧）÷（5～9）。不增加回复量没有截断；单次伤害最多 9999
敏捷战斗中行动条每帧增加「敏捷 ÷ 4」（減速时减半，加速、减速状态各增减 30%），满了就轮到行动。行动条的满值 = 100 ×（20 − 戰鬥速度），戰鬥速度在「環境設定」里设定，0～11达到满值的 4 倍后一帧就满，再高也没用：戰鬥速度 11 时为 3600，0 时为 8000
閃躲敌人攻击时掷 0～9，小于閃躲值就闪开；閃躲在 6 以下时还要先通过一次 20% 的判定，实际机率只有「閃躲 × 2%」。有人刚闪开后的下一次一定不闪；全场连续 20 次没闪开后每次有 1/3 机会闪开，连续 24 次没闪开时下一次必闪10，10 以上就是必闪（刚闪开后的那一次除外）。战斗中只取 0～255，超过 255 会绕回，例如 256 等于 0
異常中异常状态的机率（十分之几）。敌人的招式带异常状态时掷 1～10，不大于異常值就中招，所以越低越好，0 完全不会中异常0 最好，10 以上必中。变成负数时按很大的数比较，反而必中
默契没有作用：战斗和其他计算都不读取，只在升级时显示增加了多少—
反應没有作用，同默契—
抗力每个元素（火、冰、風、土、毒、光、暗、電、物）一个值，以十分之一为单位：正数多受伤害（損），−1～−10 减少伤害（抗，−10 完全无效），低于 −10 把伤害变成回复（吸）−128～127
等級普通攻击的随机加成是 1～（等級 ÷ 3 + 3），法术的随机加成也计入等級战斗胜利升级只到 60 级；某个剧情旗标（0x1BB0）打开期间只到 18 级
